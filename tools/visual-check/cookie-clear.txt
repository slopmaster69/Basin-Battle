12.2 lua function CB.after(d, f) table.insert(CB.q, { GetTime() + d, f }) end
12.2 lua function CB.log(t) if ProbeLog then ProbeLog("visual-check: bot " .. t) end end
12.2 lua function CB.hit(why) CB.log(why .. " hp " .. UnitHealth("player")) CB.say(".es in a") CB.after(0.08, function() CB.say(".es in au") end) end
12.2 lua function CB.split(s) local t = {} for p in string.gfind(s .. "\t", "([^\t]*)\t") do table.insert(t, p) end return t end

12.2 lua function CB.react(w) local now = GetTime() local seen = now CB.anAt = now CB.open = 0 if w >= 1.5 then CB.after(0.3, function() if CB.anAt == seen then CB.strafe(0.7) end end) else local d = math.max(0, w - 0.15) CB.after(d, function() if CB.anAt == seen then CB.say(".es in g") end end) CB.after(d + 0.35, function() CB.say(".es in gu") end) CB.after(w + 0.4, function() if CB.anAt == seen then CB.hit("hit back") end end) end end
12.2 lua function CB.line(l) local f = CB.split(l) local v, now = f[1], GetTime() if v == "WP" and f[5] ~= "1" then local c, s = tonumber(f[3]), tonumber(f[4]) or 0 if c == 9 then if s > 0 then CB.openAt = now + 0.3 CB.open = now + s CB.log("open " .. s) else CB.open = 0 end end elseif v == "AN" and f[11] == "mob" then CB.react(tonumber(f[4]) or 0) elseif v == "HT" then CB.lastHT = now elseif v == "LO" then CB.sawLO = true CB.locked = f[2] ~= nil and f[2] ~= "" and f[2] ~= "0x0" and f[2] ~= "0X0" and f[2] ~= "0" end end
