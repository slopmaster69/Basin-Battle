# The hero's own steps in these takes: `SV.go(x, y, short, then)` runs the hero to `short` yards
# from a spot in dashes of 0.4 s at most, each aimed from where the hero stands by then (`.es
# where`, a read), so a dash cut short by a blow costs one dash, not the run; then calls `then`.
# `SV.hit()` is the attack key pressed and let go. What strikes the hero (IM) and what the hero's
# blows do (HT) go to the take's log.
11.5 lua function SV.turn(want) local d = math.mod((want - GetActionAim()) * 57.29578 + 900, 360) - 180 if math.abs(d) > 2 then FlipCameraYaw(d) end end
11.5 lua SV.w = CreateFrame("Frame") SV.w:RegisterEvent("CHAT_MSG_SYSTEM") SV.w:SetScript("OnEvent", function() local _, _, x, y = string.find(arg1 or "", "at ([%-%d%.]+), ([%-%d%.]+),") local w = SV.want if not x or not w then return end x, y = tonumber(x), tonumber(y) local dx, dy = w[1] - x, w[2] - y local run = math.sqrt(dx * dx + dy * dy) - w[3] SV.turn(math.atan2(dy, dx)) if run <= 0.4 or GetTime() > w[5] then SV.want = nil SV.log(string.format("there, %.1f yd off, at %.1f %.1f", run, x, y)) if w[4] then w[4]() end return end local secs = math.min(run / 7, 0.4) ActionDash(secs, 7, "f") SV.after(secs + 0.03, function() if SV.want then SV.say(".es where") end end) end)
11.5 lua function SV.go(x, y, short, f) SV.want = { x, y, short, f, GetTime() + 5 } SV.log("goes") SV.say(".es where") end
11.5 lua function SV.hit() SV.say(".es in a") SV.after(0.08, function() SV.say(".es in au") end) end
11.5 lua SV.old = ElwynnSurvival.Line ElwynnSurvival.Line = function(l) local v = string.sub(l, 1, 2) if v == "IM" or v == "HT" then local t = string.gsub(l, "\t", " ") SV.log(t) end if v == "HT" then SV.lastHT = GetTime() SV.hits = SV.hits + 1 local _, _, n = string.find(l, "^HT\t(%d+)") SV.dealt = SV.dealt + (tonumber(n) or 0) elseif v == "LO" then local _, _, g = string.find(l, "^LO\t([^\t]*)") SV.locked = g ~= nil and g ~= "" and g ~= "0x0" and g ~= "0X0" and g ~= "0" end return SV.old(l) end
# `SV.fight = true`: the hero fights whoever is in front of them as a player would, by their own
# keys: the lock key when nothing is locked (every 2 s), a blow every 0.75 s, a run forward
# (0.4 s) when no blow has landed for 1.5 s.
11.5 lua SV.g = CreateFrame("Frame") SV.g:SetScript("OnUpdate", function() local now = GetTime() if not SV.fight then return end if not SV.locked and now >= SV.nextLock then SV.nextLock = now + 2 SV.say(".es in lock") end if now >= SV.nextHit then SV.nextHit = now + 0.75 SV.hit() end if now - SV.lastHT > 1.5 and now >= SV.nextDash then SV.nextDash = now + 0.55 ActionDash(0.4, 7, "f") end end)
