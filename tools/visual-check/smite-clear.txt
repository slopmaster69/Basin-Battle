12.2 lua function SB.after(d, f) table.insert(SB.q, { GetTime() + d, f }) end

12.2 lua function SB.split(s) local t = {} for p in string.gfind(s .. "\t", "([^\t]*)\t") do table.insert(t, p) end return t end
# A cut is guarded from 0.13 s before it lands to 0.07 s after: up for a parry, and down long enough
# (0.6 s) before an axe string's next cut that it parries too.
12.2 lua function SB.an(c, land) local now = GetTime() SB.chase = 0 if c == 58 or c == 18 or c == 181 then SB.busy = land + 0.45 SB.after(math.max(0, land - 0.18 - now), function() SB.say(".es in d") SB.log("roll from " .. c) end) SB.after(math.max(0, land + 0.4 - now), function() SB.dash("back in after the roll") end) else SB.busy = land + 0.3 local d = math.max(0, land - 0.13 - now) SB.after(d, function() SB.say(".es in g") end) SB.after(d + 0.2, function() SB.say(".es in gu") end) SB.log("guard " .. c) end end
12.2 lua function SB.line(l) local f = SB.split(l) local v, now = f[1], GetTime() if v == "AN" and f[11] == "mob" then SB.an(tonumber(f[3]) or 0, now + (tonumber(f[4]) or 0)) elseif v == "WP" and f[5] ~= "1" then local c, s = tonumber(f[3]) or 0, tonumber(f[4]) or 0 if c == 115 and s > 0 then if SB.rack < now - 20 then SB.rack = now end SB.kneel = now + s SB.chase = 0 SB.log("he kneels " .. s) elseif SB.kneel > now then SB.kneel = 0 SB.log("he stands") end elseif v == "HT" then SB.lastHT = now SB.hits = SB.hits + 1 SB.dealt = SB.dealt + (tonumber(f[2]) or 0) if math.mod(SB.hits, 10) == 0 then SB.log("hits " .. SB.hits .. " dealt " .. SB.dealt) end elseif v == "IM" and f[7] == "1" then SB.log("struck: " .. tostring(f[4]) .. " " .. tostring(f[5])) elseif v == "LO" then SB.sawLO = true SB.locked = f[2] ~= nil and f[2] ~= "" and f[2] ~= "0x0" and f[2] ~= "0X0" and f[2] ~= "0" end end

12.2 lua SB.f = CreateFrame("Frame") SB.f:SetScript("OnUpdate", function() local now = GetTime() local i = 1 while i <= table.getn(SB.q) do local a = SB.q[i] if now >= a[1] then table.remove(SB.q, i) a[2]() else i = i + 1 end end if not SB.on then return end if SB.sawLO and not SB.locked and now >= SB.nextLock then SB.nextLock = now + 2 SB.say(".es in lock") SB.log("lock on again") end if now < SB.busy then return end local kneeling = now < SB.kneel if kneeling and now > SB.rack + 2.5 then if not SB.waits then SB.waits = true SB.log("let him arm") end return end SB.waits = false if now >= SB.nextHit then SB.nextHit = now + (kneeling and 0.55 or 0.75) SB.hit() end local wait = 2.0 if kneeling then wait = 0.8 elseif now < SB.chase then wait = 0.6 end if now - SB.lastHT > wait and now >= SB.nextDash then SB.dash(kneeling and "close in on the kneel" or (now < SB.chase and "run after him" or "after him")) end end)
