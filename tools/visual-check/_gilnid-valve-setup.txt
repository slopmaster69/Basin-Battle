# The hero's own steps in these takes: `GV.go(x, y, short, then)` runs the hero to `short` yards
# from a spot in dashes of 0.4 s at most, each aimed from where the hero stands by then (`.es
# where`, a read), so a dash cut short by a blow costs one dash, not the run; then calls `then`.
# What strikes the hero (IM) and what the hero's blows do (HT) go to the take's log.
11.5 lua function GV.turn(want) local d = math.mod((want - GetActionAim()) * 57.29578 + 900, 360) - 180 if math.abs(d) > 2 then FlipCameraYaw(d) end end
11.5 lua GV.w = CreateFrame("Frame") GV.w:RegisterEvent("CHAT_MSG_SYSTEM") GV.w:SetScript("OnEvent", function() local _, _, x, y = string.find(arg1 or "", "at ([%-%d%.]+), ([%-%d%.]+),") local w = GV.want if not x or not w then return end x, y = tonumber(x), tonumber(y) local dx, dy = w[1] - x, w[2] - y local run = math.sqrt(dx * dx + dy * dy) - w[3] if run <= 0.4 or GetTime() > w[5] then GV.want = nil GV.log(string.format("there, %.1f yd off, at %.1f %.1f", run, x, y)) if w[4] then w[4]() end return end GV.turn(math.atan2(dy, dx)) local secs = math.min(run / 7, 0.4) ActionDash(secs, 7, "f") GV.after(secs + 0.03, function() if GV.want then GV.say(".es where") end end) end)
11.5 lua function GV.go(x, y, short, f) GV.want = { x, y, short, f, GetTime() + 4 } GV.log("goes") GV.say(".es where") end
