# The Rogue's perfect dodge (docs/rogue-roles.md, build plan 1), floor markers off, on the hero's own
# screen (so the violet corners show). A Warsong Grunt 4 yd ahead swings at a Rogue; a small player
# in the interface (RB) dodges each blow and, from what it hears, lands the dodge in the blow's
# perfect moment: the blow must land 0.04 to 0.18 sec into the dodge ([dodge] evade_from and
# perfect_to), so it aims for the middle, 0.11 sec.
#
# Why the Nov 16 runs never got one, and how this does: the bot rolled 0.12 sec before the wind-up
# ran out, but the blow lands later than the line says (the server settles it once the hero's round
# trip is up, and the client sees lines a frame late), so every roll came 0.2 to 0.36 sec before its
# blow; the roll's hop back (3 yd) had already taken the Rogue out of the grunt's 3 yd reach, so the
# blow whiffed (IM `miss`) and the dodge was never weighed. Now the bot measures, for each swing, the
# gap between the roll's own echo (RO) and the blow's impact line (IM), and moves its margin by
# the difference from 0.11 sec for the next swing (a margin for each of the short and long wind-ups).
# The proof is an `AI` line of 0.9 sec (the perfect dodge's copy) in the run's log after the IM of
# the swing: `bot PERFECT` lines. A dev setup: the controller reads what the client hears.
# A grunt's charge (0.65 sec wind-up, lunging 4.5 yd) is left alone: the roll's hop back opens the range
# it needs and the lunge then ends it short or past, a whiff the dodge never weighs. The slow overhead
# (1.0 sec) stands where it is and is the one that lands in the window most.
# Look for, with no floor marker and no word: the grunt's own wind-up; the Rogue's body flickering
# see-through for 0.5 sec with a copy (alpha 0.85, standing 0.9 sec) left where the blow fell;
# Vanish's puff and sound; the short stop and violet corners (1.0 sec) on the screen.

4.3 lua ElwynnSurvivalSettings = ElwynnSurvivalSettings or {}; ElwynnSurvivalSettings.noMarks = 1; if ElwynnSurvival and ElwynnSurvival.ClearMarks then ElwynnSurvival.ClearMarks() end
4.4 lua RB = { q = {}, on = false, n = 0, most = 30, want = 0.11, ms = -0.04, ml = 0.0, perfect = 0, frames = 0, last = GetTime() }
4.4 lua function RB.say(t) SendChatMessage(t, "SAY") end
4.4 lua function RB.after(d, f) table.insert(RB.q, { GetTime() + d, f }) end
4.4 lua function RB.log(t) if ProbeLog then ProbeLog("visual-check: bot " .. t) end end
4.4 lua function RB.split(s) local t = {} for p in string.gfind(s .. "\t", "([^\t]*)\t") do table.insert(t, p) end return t end
4.4 lua function RB.react(w) if not RB.on or w <= 0 or RB.n >= RB.most then return end if w > 0.62 and w < 0.68 then RB.log("charge, w=" .. w .. ": not dodged (a roll back opens the range and the blow whiffs)") return end RB.n = RB.n + 1 local long = w >= 0.9 RB.cls = long local m = long and RB.ml or RB.ms RB.log("AN w=" .. w .. " m=" .. string.format("%.3f", m) .. " n=" .. RB.n) RB.after(math.max(0, w - m), function() RB.say(".es in d") RB.log("dodge sent n=" .. RB.n) end) end
4.4 lua function RB.pair() if RB.rot and RB.imt and math.abs(RB.rot - RB.imt) < 0.7 then local d = RB.imt - RB.rot local adj = (RB.want - d) * 0.8 if adj > 0.12 then adj = 0.12 elseif adj < -0.12 then adj = -0.12 end if RB.cls then RB.ml = RB.ml + adj else RB.ms = RB.ms + adj end RB.log("gap " .. string.format("%.3f", d) .. " (roll to impact); margins short " .. string.format("%.3f", RB.ms) .. " long " .. string.format("%.3f", RB.ml)) RB.rot = nil RB.imt = nil end end
4.4 lua function RB.line(l) local f = RB.split(l) if f[1] == "AN" and f[11] == "mob" then RB.react(tonumber(f[4]) or 0) elseif f[1] == "RO" and f[4] == "1" then RB.me = f[2] RB.rot = GetTime() RB.pair() elseif f[1] == "IM" and RB.me ~= nil and f[3] == RB.me then RB.imt = GetTime() RB.log("IM " .. f[5]) RB.pair() elseif f[1] == "AI" then RB.log("AI line " .. (f[3] or "?")) if f[3] ~= nil and tonumber(f[3]) and tonumber(f[3]) > 0 and tonumber(f[3]) < 1.5 then RB.perfect = RB.perfect + 1 RB.log("PERFECT " .. RB.perfect .. " (copy " .. f[3] .. " sec)") end end end
4.4 lua local old = ElwynnSurvival.Line ElwynnSurvival.Line = function(l) RB.line(l) return old(l) end
4.4 lua RB.f = CreateFrame("Frame") RB.f:SetScript("OnUpdate", function() local now = GetTime() RB.frames = RB.frames + 1 if now - RB.last >= 3 then RB.log("frames/s " .. string.format("%.1f", RB.frames / (now - RB.last))) RB.frames = 0 RB.last = now end local i = 1 while i <= table.getn(RB.q) do local a = RB.q[i] if now >= a[1] then table.remove(RB.q, i) a[2]() else i = i + 1 end end end)
