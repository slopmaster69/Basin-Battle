12.4 lua function RB.say(t) SendChatMessage(t, "SAY") end
12.4 lua function RB.after(d, f) table.insert(RB.q, { GetTime() + d, f }) end

12.4 lua function RB.dash(why, keys, secs) RB.nextDash = GetTime() + 0.55 ActionDash(secs or 0.4, 7, keys or "f") RB.log(why) end
12.4 lua function RB.split(s) local t = {} for p in string.gfind(s .. "\t", "([^\t]*)\t") do table.insert(t, p) end return t end
12.4 lua function RB.turn(want) local d = math.mod((want - GetActionAim()) * 57.29578 + 900, 360) - 180 if math.abs(d) > 2 then FlipCameraYaw(d) end end
12.4 lua function RB.hold(a, b) table.insert(RB.b, { a, b }) end
12.4 lua function RB.busy(now) local i, busy = 1, false while i <= table.getn(RB.b) do local w = RB.b[i] if now >= w[2] then table.remove(RB.b, i) else if now >= w[1] then busy = true end i = i + 1 end end return busy end
# His blows (AN: the clip, its take, when it lands). Rolled: the overhead (58), the stomp and the
# pound (54), the roar (55), the rush (18, take 0). Guarded as they land: the swings (18, 17, 59).
# The hero's own blows stop 0.8 s before his lands. In the last stand (after "SMASH"), a step
# round him after each blow, off the burning iron.
12.4 lua function RB.an(c, take, land) local now = GetTime() if RB.run then return end if c == 58 or c == 54 or c == 55 or (c == 18 and take == 0) then RB.hold(land - 0.8, land + 0.45) RB.after(math.max(0, land - 0.18 - now), function() RB.say(".es in d") RB.log("roll from " .. c) end) RB.after(math.max(0, land + 0.4 - now), function() if not RB.run then RB.dash("back in after the roll") end end) else RB.hold(land - 0.8, land + 0.3) local d = math.max(0, land - 0.13 - now) RB.after(d, function() RB.say(".es in g") end) RB.after(d + 0.2, function() RB.say(".es in gu") end) RB.log("guard " .. c) end if RB.stand then RB.after(math.max(0, land + 0.85 - now), function() if not RB.run then RB.dash("round him, off the iron", "l", 0.8) end end) end end
# A fall or a blast warned with no creature's blow (WN, no guid): rolled as it lands.
12.4 lua function RB.wn(s) local now = GetTime() if RB.run or s < 0.5 then return end RB.hold(now + s - 0.8, now + s + 0.45) RB.after(math.max(0, s - 0.18), function() RB.say(".es in d") RB.log("roll from a fall") end) RB.after(s + 0.4, function() if not RB.run then RB.dash("back in after the fall") end end) end
12.4 lua function RB.line(l) local f = RB.split(l) local v, now = f[1], GetTime() if v == "AN" and f[11] == "mob" then RB.an(tonumber(f[3]) or 0, tonumber(f[8]) or 0, now + (tonumber(f[4]) or 0)) elseif v == "WN" and (f[3] == nil or f[3] == "") then RB.wn(tonumber(f[2]) or 0) elseif v == "HT" then RB.lastHT = now RB.hits = RB.hits + 1 RB.dealt = RB.dealt + (tonumber(f[2]) or 0) RB.log("lands " .. RB.hits .. " of " .. RB.tries .. ": " .. tostring(f[2]) .. " " .. tostring(f[4]) .. ", dealt " .. RB.dealt) elseif v == "IM" and f[7] == "1" then RB.struck = RB.struck + 1 if f[5] == "hit" then RB.caught = RB.caught + 1 end RB.log("struck " .. RB.struck .. " caught " .. RB.caught .. ": " .. tostring(f[4]) .. " " .. tostring(f[5])) elseif v == "LO" then RB.sawLO = true RB.locked = f[2] ~= nil and f[2] ~= "" and f[2] ~= "0x0" and f[2] ~= "0X0" and f[2] ~= "0" if RB.locked then RB.miss = 0 end end end

# The collapse: from his yell, 0.2 to 0.3 s after each later section's warning begins, the hero
# lets go of him and runs to floor past that section's line (where it is read from `.es where`,
# which changes nothing), then turns back the way it came and locks on again (a lock that does
# not take is tried again every 2 s, looking another way). Once he is dead the same steering walks the hero
# up to the last face and stops there, to show it standing.
12.4 lua function RB.go(w, look) RB.run = w RB.wi = 1 RB.nextWhere = 0 RB.look = look if not look then RB.say(".es in unlock") end RB.log(look and "walks to the last face" or "runs from the collapse") end
12.4 lua RB.w = CreateFrame("Frame") RB.w:RegisterEvent("CHAT_MSG_SYSTEM") RB.w:SetScript("OnEvent", function() local _, _, x, y = string.find(arg1 or "", "at ([%-%d%.]+), ([%-%d%.]+),") if not x or not RB.run then return end x, y = tonumber(x), tonumber(y) local w, n = RB.run[RB.wi], RB.run[RB.wi + 1] local dx, dy = w[1] - x, w[2] - y local d = math.sqrt(dx * dx + dy * dy) if n then local ex, ey = n[1] - x, n[2] - y local e = math.sqrt(ex * ex + ey * ey) local gx, gy = n[1] - w[1], n[2] - w[2] if d < 1.5 or e < math.sqrt(gx * gx + gy * gy) then RB.wi = RB.wi + 1 dx, dy, d = ex, ey, e end end if d < 1.5 then local b = RB.run[1] RB.run = nil RB.log("stands again at " .. x .. " " .. y) if RB.look then RB.turn(0) else RB.turn(math.atan2(b[2] - y, b[1] - x)) RB.after(0.3, function() RB.say(".es in lock") end) RB.nextLock = GetTime() + 2.3 RB.lastHT = GetTime() end return end RB.turn(math.atan2(dy, dx)) ActionDash(math.min(d / 7, 0.8), 7, "f") end)
